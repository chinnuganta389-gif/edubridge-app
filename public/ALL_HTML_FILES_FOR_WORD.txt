    // Load Featured Resources on Homepage
    async function loadFeaturedResources() {
      const grid = document.getElementById('featuredResourcesGrid');
      if (!grid) return;

      try {
        const res = await fetch('/api/resources?featured=1&limit=6');
        const data = await res.json();

        if (data.resources && data.resources.length > 0) {
          grid.innerHTML = data.resources.map(r => EduApp.renderResourceCard(r)).join('');
        } else {
          grid.innerHTML = '<p style="grid-column: 1/-1; text-align: center; color: var(--gray-500);">Loading educational platforms...</p>';
        }
      } catch (err) {
        grid.innerHTML = '<p style="grid-column: 1/-1; text-align: center; color: #ef4444;">Failed to load resources. Please ensure the backend server is running.</p>';
      }
    }

    document.addEventListener('DOMContentLoaded', () => {
      loadFeaturedResources();
    });

    window.addEventListener('languageChanged', () => {
      loadFeaturedResources();
    });
  </script>

    let searchTimeout = null;
    let selectedSubcategory = '';

    function getQueryParams() {
      const params = new URLSearchParams(window.location.search);
      return {
        search: params.get('search') || '',
        category: params.get('category') || 'all',
        skill_level: params.get('skill_level') || 'all',
        pricing: params.get('pricing') || 'all',
        subcategory: params.get('subcategory') || ''
      };
    }

    function initFromURL() {
      const q = getQueryParams();
      if (q.search) document.getElementById('searchInput').value = q.search;
      if (q.category) document.getElementById('filterCategory').value = q.category;
      if (q.skill_level) document.getElementById('filterSkillLevel').value = q.skill_level;
      if (q.pricing) document.getElementById('filterPricing').value = q.pricing;
      if (q.subcategory) {
        selectedSubcategory = q.subcategory;
        updateActiveTopicBadge(selectedSubcategory);
      }
    }

    function debounceSearch() {
      clearTimeout(searchTimeout);
      searchTimeout = setTimeout(() => {
        fetchResources();
      }, 350);
    }

    function setTopicFilter(topic) {
      selectedSubcategory = topic;
      updateActiveTopicBadge(topic);
      fetchResources();
    }

    function updateActiveTopicBadge(topic) {
      const badge = document.getElementById('activeTopicBadge');
      const text = document.getElementById('activeTopicText');
      if (topic) {
        text.textContent = topic;
        badge.style.display = 'inline-block';
      } else {
        badge.style.display = 'none';
      }
    }

    function resetFilters() {
      document.getElementById('searchInput').value = '';
      document.getElementById('filterCategory').value = 'all';
      document.getElementById('filterSkillLevel').value = 'all';
      document.getElementById('filterPricing').value = 'all';
      document.getElementById('sortSelect').value = 'popular';
      selectedSubcategory = '';
      updateActiveTopicBadge('');
      fetchResources();
    }

    async function fetchResources() {
      const grid = document.getElementById('exploreResourcesGrid');
      const countLabel = document.getElementById('resultsCount');

      const search = document.getElementById('searchInput').value.trim();
      const category = document.getElementById('filterCategory').value;
      const skill_level = document.getElementById('filterSkillLevel').value;
      const pricing = document.getElementById('filterPricing').value;
      const sort = document.getElementById('sortSelect').value;

      const params = new URLSearchParams();
      if (search) params.append('search', search);
      if (category && category !== 'all') params.append('category', category);
      if (skill_level && skill_level !== 'all') params.append('skill_level', skill_level);
      if (pricing && pricing !== 'all') params.append('pricing', pricing);
      if (selectedSubcategory) params.append('subcategory', selectedSubcategory);
      if (sort) params.append('sort', sort);

      try {
        const headers = {};
        const token = localStorage.getItem('edubridge_token');
        if (token) headers['Authorization'] = `Bearer ${token}`;

        const res = await fetch(`/api/resources?${params.toString()}`, { headers });
        const data = await res.json();

        if (data.resources && data.resources.length > 0) {
          countLabel.innerHTML = `Showing <strong>${data.resources.length}</strong> educational platform${data.resources.length > 1 ? 's' : ''}`;
          grid.innerHTML = data.resources.map(r => EduApp.renderResourceCard(r)).join('');
        } else {
          countLabel.textContent = '0 platforms found';
          grid.innerHTML = `
            <div style="grid-column: 1/-1; text-align: center; padding: 3rem 1.5rem; background: var(--white); border-radius: var(--radius-lg); border: 1px solid var(--gray-200);">
              <span style="font-size: 2.5rem; display: block; margin-bottom: 0.75rem;">🔍</span>
              <h3 style="font-size: 1.25rem; font-weight: 700; color: var(--gray-800); margin-bottom: 0.5rem;">No educational platforms found</h3>
              <p style="font-size: 0.92rem; color: var(--gray-600); max-width: 500px; margin: 0 auto 1.5rem;">
                We couldn't find any resources matching your search criteria. Try removing filters or searching for terms like "Python", "Java", "DSA", or "NPTEL".
              </p>
              <button onclick="resetFilters()" class="btn btn-primary btn-sm">Reset All Filters</button>
            </div>
          `;
        }
      } catch (err) {
        countLabel.textContent = 'Error loading resources';
        grid.innerHTML = '<p style="grid-column:1/-1; color:red; text-align:center;">Failed to connect to the server.</p>';
      }
    }

    document.addEventListener('DOMContentLoaded', () => {
      initFromURL();
      fetchResources();
    });

    window.addEventListener('languageChanged', () => {
      fetchResources();
    });

    let currentPracticeFilter = 'All';

    async function loadPracticeResources() {
      const grid = document.getElementById('codingPracticeGrid');
      grid.innerHTML = '<p style="grid-column:1/-1; text-align:center;">Loading coding practice platforms...</p>';

      try {
        const headers = {};
        const token = localStorage.getItem('edubridge_token');
        if (token) headers['Authorization'] = `Bearer ${token}`;

        let url = '/api/resources?category=coding_practice';
        if (currentPracticeFilter !== 'All') {
          if (['Beginner', 'Intermediate', 'Advanced'].includes(currentPracticeFilter)) {
            url += `&skill_level=${encodeURIComponent(currentPracticeFilter)}`;
          } else {
            url += `&subcategory=${encodeURIComponent(currentPracticeFilter)}`;
          }
        }

        const res = await fetch(url, { headers });
        const data = await res.json();

        if (data.resources && data.resources.length > 0) {
          grid.innerHTML = data.resources.map(r => EduApp.renderResourceCard(r)).join('');
        } else {
          grid.innerHTML = `
            <div style="grid-column:1/-1; text-align:center; padding:2rem; background:#fff; border-radius:12px;">
              <p>No platforms matching filter. Showing all practice platforms:</p>
              <button onclick="filterPractice('All', document.querySelector('.p-btn'))" class="btn btn-primary btn-sm" style="margin-top:0.5rem;">Show All</button>
            </div>
          `;
        }
      } catch (err) {
        grid.innerHTML = '<p style="grid-column:1/-1; color:red; text-align:center;">Failed to load resources.</p>';
      }
    }

    function filterPractice(filter, btn) {
      currentPracticeFilter = filter;
      document.querySelectorAll('.p-btn').forEach(b => b.classList.remove('active'));
      if (btn) btn.classList.add('active');
      loadPracticeResources();
    }

    document.addEventListener('DOMContentLoaded', () => {
      loadPracticeResources();
    });

    window.addEventListener('languageChanged', () => {
      loadPracticeResources();
    });
  </script>

    async function loadSubjectPlatforms(subjectName) {
      const grid = document.getElementById('subjectResourcesGrid');
      grid.innerHTML = '<p style="grid-column:1/-1; text-align:center;">Loading recommended platforms...</p>';

      try {
        const headers = {};
        const token = localStorage.getItem('edubridge_token');
        if (token) headers['Authorization'] = `Bearer ${token}`;

        // Fetch resources for this subject
        const res = await fetch(`/api/resources?category=btech_subjects&subcategory=${encodeURIComponent(subjectName)}`, { headers });
        const data = await res.json();

        if (data.resources && data.resources.length > 0) {
          grid.innerHTML = data.resources.map(r => EduApp.renderResourceCard(r)).join('');
        } else {
          // If no specific match, show all btech subject platforms
          const allRes = await fetch(`/api/resources?category=btech_subjects`, { headers });

          grid.innerHTML = allData.resources.map(r => EduApp.renderResourceCard(r)).join('');

    async function loadAptitudeResources() {
      const grid = document.getElementById('aptitudeGrid');
      grid.innerHTML = '<p style="grid-column:1/-1; text-align:center;">Loading aptitude resources...</p>';

      try {
        const headers = {};
        const token = localStorage.getItem('edubridge_token');
        if (token) headers['Authorization'] = `Bearer ${token}`;

        const res = await fetch('/api/resources?category=aptitude', { headers });
        const data = await res.json();

        if (data.resources && data.resources.length > 0) {
          grid.innerHTML = data.resources.map(r => EduApp.renderResourceCard(r)).join('');
        } else {
          grid.innerHTML = '<p style="grid-column:1/-1; text-align:center;">No resources found.</p>';
        }
      } catch (err) {
        grid.innerHTML = '<p style="grid-column:1/-1; color:red; text-align:center;">Failed to load resources.</p>';
      }
    }

    document.addEventListener('DOMContentLoaded', () => {
      loadAptitudeResources();
    });

      loadAptitudeResources();

    async function loadDashboardData() {
      const token = localStorage.getItem('edubridge_token');
      if (!token) {
        window.location.href = 'login.html';
        return;
      }

      try {
        const res = await fetch('/api/dashboard/stats', {
          headers: { 'Authorization': `Bearer ${token}` }
        });
        const data = await res.json();

        if (!res.ok) {
          throw new Error(data.error || 'Failed to load profile');
        }

        // Populate User Info
        document.getElementById('dashStudentName').textContent = data.user.name;
        document.getElementById('dashStudentMeta').textContent = `${data.user.email} • ${data.user.study_level || 'Student'}`;
        document.getElementById('dashVillageName').textContent = data.user.village || 'Rural Village Learner';

        // Stats
        document.getElementById('statSavedCount').textContent = data.stats.savedBookmarks;
        document.getElementById('statVisitsCount').textContent = data.stats.visitedPlatforms;
        document.getElementById('statInterestsCount').textContent = data.stats.learningInterestsCount;

        // Interests List
        const interestsContainer = document.getElementById('dashInterestsList');
        if (data.user.interests && data.user.interests.length > 0) {
          interestsContainer.innerHTML = data.user.interests.map(i => `
            <span class="badge badge-beginner" style="font-size:0.8rem; padding:0.3rem 0.75rem;">✔ ${i}</span>
          `).join('');
        } else {
          interestsContainer.innerHTML = '<span style="font-size:0.85rem; color:#64748b;">No interests selected yet. Click "Edit Profile" above to personalize your recommendations.</span>';
        }

        // Bookmarks Grid
        const bookmarksGrid = document.getElementById('dashBookmarksGrid');
        if (data.bookmarks && data.bookmarks.length > 0) {
          bookmarksGrid.innerHTML = data.bookmarks.map(b => EduApp.renderResourceCard({ ...b, is_bookmarked: true })).join('');
        } else {
          bookmarksGrid.innerHTML = `
            <div style="grid-column:1/-1; text-align:center; padding:2rem; background:#fff; border-radius:12px; border:1px solid #e2e8f0;">
              <p style="color:#64748b; font-size:0.95rem;">You haven't bookmarked any learning platforms yet.</p>
              <a href="explore.html" class="btn btn-primary btn-sm" style="margin-top:0.75rem;">Explore & Bookmark Websites</a>
            </div>
          `;
        }

        // Recently Visited Platforms
        const visitedList = document.getElementById('dashVisitedList');
        if (data.recentVisits && data.recentVisits.length > 0) {
          visitedList.innerHTML = data.recentVisits.map(v => {
            const date = new Date(v.last_visited).toLocaleDateString('en-IN', { day:'numeric', month:'short', hour:'2-digit', minute:'2-digit' });
            return `
              <div class="visited-item">
                <div class="visited-info">
                  <div style="width:36px; height:36px; border-radius:6px; background:#eff6ff; display:flex; align-items:center; justify-content:center; font-weight:700; color:#1d4ed8;">
                    ${v.name.charAt(0)}
                  </div>
                  <div>
                    <strong style="color:var(--gray-900); font-size:0.92rem;">${v.name}</strong>
                    <div style="font-size:0.75rem; color:var(--gray-500);">${v.category.replace('_', ' ')} • Visited: ${date}</div>
                  </div>
                </div>
                <button onclick="EduApp.openPlatform(${v.id}, '${v.url}')" class="btn btn-outline btn-sm">
                  Launch ↗
                </button>
              </div>
            `;
          }).join('');
        } else {
          visitedList.innerHTML = '<p style="color:#64748b; font-size:0.88rem; background:#fff; padding:1rem; border-radius:8px; border:1px solid #e2e8f0;">No recent visits recorded. When you click "OPEN PLATFORM", visits will be logged here for quick resumption.</p>';
        }

        // Recommendations Grid
        const recsGrid = document.getElementById('dashRecommendedGrid');
        if (data.recommendations && data.recommendations.length > 0) {
          recsGrid.innerHTML = data.recommendations.map(r => EduApp.renderResourceCard(r)).join('');
        }
      } catch (err) {
        console.error(err);
        EduApp.toast('Please login to view your personal dashboard.', 'error');
        setTimeout(() => window.location.href = 'login.html', 1000);
      }
    }

    function openProfileModal() {
      if (!window.EduAuth.currentUser) return;
      const u = window.EduAuth.currentUser;
      document.getElementById('editName').value = u.name || '';
      document.getElementById('editVillage').value = u.village || '';
      document.getElementById('editCollege').value = u.college_school || '';
      document.getElementById('editLanguage').value = u.preferred_language || 'en';

      const userInterests = Array.isArray(u.interests) ? u.interests : [];
      document.querySelectorAll('#interestCheckboxes input').forEach(cb => {
        cb.checked = userInterests.includes(cb.value);
      });

      document.getElementById('profileModal').classList.add('active');
    }

    function closeProfileModal() {
      document.getElementById('profileModal').classList.remove('active');
    }

    async function handleProfileUpdate(e) {
      e.preventDefault();
      const name = document.getElementById('editName').value.trim();
      const village = document.getElementById('editVillage').value.trim();
      const college_school = document.getElementById('editCollege').value.trim();
      const preferred_language = document.getElementById('editLanguage').value;

      const selectedInterests = [];
      document.querySelectorAll('#interestCheckboxes input:checked').forEach(cb => {
        selectedInterests.push(cb.value);
      });

      const res = await window.EduAuth.updateProfile({
        name,
        village,
        college_school,
        preferred_language,
        interests: selectedInterests
      });

      if (res && res.success) {
        EduApp.toast('Profile & learning preferences updated!', 'success');
        closeProfileModal();
        loadDashboardData();
      } else {
        EduApp.toast('Failed to update profile.', 'error');
      }
    }

    document.addEventListener('DOMContentLoaded', () => {
      loadDashboardData();
    });

    window.addEventListener('languageChanged', () => {
      loadDashboardData();
    });

      <!-- 1. Problem Statement -->
      <article class="about-card">
        <h3>1. Problem Statement</h3>
        <p style="font-size:0.95rem; color:var(--gray-700); line-height:1.7; margin-bottom:1rem;">
          In contemporary India, the expansion of mobile broadband (4G/5G) has placed smartphones in almost every rural household. However, a profound <strong>"Awareness Divide"</strong> persists. While urban students frequently utilize platforms such as W3Schools, GeeksforGeeks, HackerRank, LeetCode, NPTEL, and Coursera to build high-demand technical capabilities and secure employment, rural village students remain predominantly unaware of these free global opportunities.
        </p>
        <p style="font-size:0.95rem; color:var(--gray-700); line-height:1.7;">
          Due to the absence of technical mentors, high fees charged by commercial coaching institutes, and language hesitation, rural learners often misinterpret the internet merely as an entertainment or social media vehicle rather than a transformative vehicle for career advancement.
        </p>
      </article>

      <!-- 2. Challenges Faced by Rural Students -->
      <article class="about-card">
        <h3>2. Challenges Faced by Rural Students</h3>
        <div style="display:grid; grid-template-columns:repeat(auto-fit, minmax(260px, 1fr)); gap:1.25rem; margin-top:1rem;">
          <div style="background:var(--gray-50); padding:1rem; border-radius:8px; border:1px solid var(--gray-200);">
            <strong style="color:var(--accent-700); display:block; margin-bottom:0.25rem;">Lack of Awareness</strong>
            <p style="font-size:0.85rem; color:var(--gray-600);">Unawareness of existing free resources like freeCodeCamp, CS50 Harvard, and Govt. of India SWAYAM portals.</p>
          </div>
          <div style="background:var(--gray-50); padding:1rem; border-radius:8px; border:1px solid var(--gray-200);">
            <strong style="color:var(--accent-700); display:block; margin-bottom:0.25rem;">Language Hesitation</strong>
            <p style="font-size:0.85rem; color:var(--gray-600);">Fear of fast, dense academic English and lack of regional language (Telugu) orientation for beginners.</p>
          </div>
          <div style="background:var(--gray-50); padding:1rem; border-radius:8px; border:1px solid var(--gray-200);">
            <strong style="color:var(--accent-700); display:block; margin-bottom:0.25rem;">Hardware Misconception</strong>
            <p style="font-size:0.85rem; color:var(--gray-600);">Belief that programming requires an expensive PC, unaware that modern in-browser editors run on mobile phones.</p>
          </div>
          <div style="background:var(--gray-50); padding:1rem; border-radius:8px; border:1px solid var(--gray-200);">
            <strong style="color:var(--accent-700); display:block; margin-bottom:0.25rem;">Lack of Roadmap & Mentorship</strong>
            <p style="font-size:0.85rem; color:var(--gray-600);">Confusion regarding what to learn first, which order to follow, and which certifications employers value.</p>
          </div>
        </div>
      </article>

      <!-- 3. Objectives of the Project -->
      <article class="about-card">
        <h3>3. Objectives of the Study & EduBridge</h3>
        <ul style="padding-left:1.5rem; font-size:0.95rem; color:var(--gray-700); line-height:1.8;">
          <li>To survey and identify digital education awareness levels among rural village students.</li>
          <li>To design and deploy a centralized, single-point web application (<strong>EduBridge</strong>) that organizes authentic, free educational platforms.</li>
          <li>To provide beginner-friendly digital literacy guidelines in simple English and Telugu (తెలుగు) so first-time users can navigate safely.</li>
          <li>To organize university B.Tech engineering curriculum subjects with curated free video lectures and revision notes.</li>
          <li>To eliminate financial anxiety by transparently categorizing platforms into 100% Free, Free Audit, and Govt-sponsored offerings.</li>
          <li>To establish clear step-by-step career roadmaps for roles such as Full Stack Developer, Python Developer, and Data Analyst.</li>
        </ul>
      </article>

      <!-- 4. How EduBridge Helps Students -->
      <article class="about-card">
        <h3>4. How EduBridge Helps Students</h3>
        <p style="font-size:0.95rem; color:var(--gray-700); line-height:1.7; margin-bottom:1rem;">
          EduBridge functions as an <strong>Educational Gateway and Discovery Hub</strong>. Rather than forcing students to navigate through search engine clutter or falling prey to commercial marketing ads:
        </p>
        <div style="background:#eff6ff; border-left:4px solid var(--primary-600); padding:1.25rem; border-radius:0 8px 8px 0; font-size:0.92rem; color:var(--primary-900); line-height:1.7;">
          ✓ A rural student opens EduBridge on any mobile phone or browser.<br>
          ✓ They choose their subject (e.g. Python, Operating Systems, or TCS Placement Aptitude).<br>
          ✓ They read an intuitive summary in English or Telugu explaining what the platform does.<br>
          ✓ They click <strong>"OPEN PLATFORM"</strong> and are directly launched into the genuine verified website (W3Schools, HackerRank, NPTEL) in a new browser tab.<br>
          ✓ They can save platforms to their personal dashboard bookmarks and resume their learning anytime.
        </div>
